async function updateDashboard(){ const container = document.getElementById("cards-container"); const teamResults = await Promise.all(TEAMS.map(t => getTeamData(t))); // ✅ Hide teams if OFF DAY and next game is > 3 days away const HIDE_AFTER_DAYS = 3; const now = Date.now(); const visible = teamResults .map((data, idx) => ({ data, idx })) .filter(({ data }) => { // Always show if game exists today in any state if (data.status === "in" || data.status === "post" || data.status === "pre") return true; // If we don't know next game time, keep it visible if (!data.startIso) return true; // If next game is soon enough, keep it visible const diffDays = (new Date(data.startIso).getTime() - now) / 86400000; return diffDays <= HIDE_AFTER_DAYS; }); // ✅ Set number of columns to match visible cards const cols = Math.max(1, Math.min(4, visible.length || 4)); document.documentElement.style.setProperty("--cols", cols); // Render container.innerHTML = ""; // If somehow everything got filtered out, just show all 4 const listToRender = visible.length ? visible : teamResults.map((data, idx) => ({ data, idx })); listToRender.forEach(({ data, idx: i }) => { const card = document.createElement("div"); card.className = "card"; const badgeClass = `status-badge status-${data.status}`; const opponentBlock = (atOrVs, oppLogo, oppName, timeText, countdown) => { const hasLogo = !!oppLogo; return `
NEXT
${atOrVs}
${hasLogo ? `` : `
${oppName}
`}
${countdown ? `
${countdown}
` : ``}
${timeText}
`; }; const scoreBlock = () => { const pillText = normalizeDetailForPill(data.detail); const alertsHtml = (data.alerts && data.alerts.length) ? `
${data.alerts.map(a => `
${a}
`).join("")}
` : ""; const myBattingClass = data.myBatting ? "batting" : ""; const oppBattingClass = data.oppBatting ? "batting" : ""; return `
${data.myScore}
-
${data.oppScore}
${alertsHtml} ${pillText ? `
${pillText}
` : ``} `; }; let bottomBlock = ""; if (data.status === "in" || data.status === "post") bottomBlock = scoreBlock(); else if (data.status === "pre") bottomBlock = opponentBlock(data.atOrVs, data.oppLogo, data.oppName, data.detail, data.countdown); else bottomBlock = opponentBlock(data.nextAtOrVs, data.nextOppLogo, data.nextOpponent, data.nextTime, data.countdown); card.innerHTML = `
${statusLabel(data.status)}
${TEAMS[i].name}
${data.record}
${data.standings ? `
${data.standings}
` : ``}
${bottomBlock} `; container.appendChild(card); }); }
